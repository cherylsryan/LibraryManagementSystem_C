// file handling functions
void saveMembers(){
    ofstream file("members.txt");

    if (!file){
        cout << "\nError: Unable to save member data.\n";
        return;
    }

    for (const Member& member : members){
        file << member.memberID << "|" << member.memberName << "|"
            << member.phoneNum << "|" << member.email << "|"
            << member.membershipType << "|" << member.membershipStatus << endl;
    }
    file.close();
}

void loadMembers(){
    ifstream file("members.txt");

    if (!file) {
        return;
    }

    members.clear();

    string line;
    while (getline(file, line)) {
        if (line.empty()) {
            continue;
        }

        Member member;
        stringstream ss(line);

        getline(ss, member.memberID, '|');
        getline(ss, member.memberName, '|');
        getline(ss, member.phoneNum, '|');
        getline(ss, member.email, '|');
        getline(ss, member.membershipType, '|');
        getline(ss, member.membershipStatus);

        members.push_back(member);
    }

    file.close();
}

bool validName(const string& name){
    if (name.empty()){
        return false;
    }

    for (char c : name){
        if (!isalpha(c) && c != ' '){
            return false;
        }
    }

    return true;
}

bool validEmail(const string& email){
    int atCount = 0;
    int dotCount = 0;
    int atPosition = -1;
    int dotPosition = -1;

    for (int i = 0; i < email.length(); i++){
        if (email[i] == '@'){
            atCount++;
            atPosition = i;
        }

        if (email[i] == '.'){
            dotCount++;
            dotPosition = i;
        }
    }

    if (email.empty()){
        return false;
    }

    if (atCount != 1){
        return false;
    }

    if (dotCount < 1){
        return false;
    }

    if (atPosition == 0){
        return false;
    }

    if (dotPosition <= atPosition + 1){
        return false;
    }
    return true;
}

bool validPhone(const string& phone){
    // Check if empty
    if (phone.empty()){
        return false;
    }

    // Check if all characters are digits
    for (char c : phone){
        if (!isdigit(c)){
            return false;
        }
    }

    // Check length
    if (phone.length() < 10 || phone.length() > 11){
        return false;
    }
    return true;
}

string toUpperCase(string text){
    for (char& c : text)
    {
        c = toupper(c);
    }
    return text;
}

}

string generateMemberID(){
    int highestID = 0;
    for (const Member& member : members)
    {
        if (member.memberID.length() > 1 &&
            member.memberID[0] == 'M')
        {
            try
            {
                int currentID = stoi(member.memberID.substr(1));

                if (currentID > highestID)
                {
                    highestID = currentID;
                }
            }
            catch (...)
            {
                // Ignore invalid Member ID
            }
        }
    }

    highestID++;

    string memberID = "M";
    if (highestID < 10)
    {
        memberID += "00" + to_string(highestID);
    }
    else if (highestID < 100)
    {
        memberID += "0" + to_string(highestID);
    }
    else
    {
        memberID += to_string(highestID);
    }
    return memberID;
}

// delete member function
void deleteMember(){
    while (true)
    {
        cout << "\n=============================================\n";
        cout << "              DELETE MEMBER\n";
        cout << "=============================================\n";

        string searchID;

        cout << "Enter Member ID (or 0 to cancel): ";
        cin >> searchID;
        searchID = toUpperCase(searchID);

        if (searchID == "0")
        {
            cout << "\nDelete cancelled.\n";
            return;
        }

        bool found = false;

        for (int i = 0; i < members.size(); i++)
        {
            if (members[i].memberID == searchID)
            {
                found = true;

                cout << "\nMember found!\n";
                cout << "---------------------------------------------\n";
                cout << "Member ID       : " << members[i].memberID << endl;
                cout << "Member Name     : " << members[i].memberName << endl;
                cout << "Membership Type : " << members[i].membershipType << endl;
                cout << "Status          : " << members[i].membershipStatus << endl;
                cout << "---------------------------------------------\n";

                char confirmation;

                cout << "Delete this member? (Y/N): ";
                cin >> confirmation;

                if (confirmation == 'Y' || confirmation == 'y'){
                    members.erase(members.begin() + i);
                    saveMembers();
                    cout << "\nMember deleted successfully!\n";
                }
                else{
                    cout << "\nDelete cancelled.\n";
                }
                return;
            }
        }

        // If ID was not found
        if (!found){
            cout << "\nMember with ID " << searchID << " not found.\n";
            cout << "Please enter a valid Member ID.\n";
        }
    }
}

// login function to authenticate users
bool login(string& loggedInRole,string& loggedInMemberID)
{
    string username;
    string password;

    cout << "Welcome to the Smart Library Management System!\n";
    cout << "Please log in to continue.\n\n";

    cout << "Username: ";
    cin >> username;

    cout << "Password: ";
    cin >> password;

    ifstream file("users.txt");

    if (!file)
    {
        cout << "\nError: Unable to open users.txt\n";
        return false;
    }

    string line;

    while (getline(file, line)) {
        if (line.empty())
        {
            continue;
        }

        string storedUsername;
        string storedPassword;
        string storedRole;
        string storedMemberID;

        stringstream ss(line);

        getline(ss, storedUsername, '|');
        getline(ss, storedPassword, '|');
        getline(ss, storedRole, '|');
        getline(ss, storedMemberID);

        // Check login
        if (username == storedUsername && password == storedPassword)
        {
            loggedInRole = storedRole;
            loggedInMemberID = toUpperCase(storedMemberID);

            file.close();

            cout << "\nLogin successful!\n";
            cout << "Welcome " << username << "!\n";

            pauseScreen();

            return true;
        }
    }

    file.close();

    cout << "\nInvalid username or password.\n";
    cout << "Please check your credentials again.\n\n";
    return false;
}
